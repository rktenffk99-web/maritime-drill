// 기본 해설도 허용 태그/스타일만 남겨 innerHTML 주입을 차단합니다.
function sanitizeExplainHtml(input){
  if(typeof input!=='string') return '';
  const template=document.createElement('template');
  template.innerHTML=input;
  const allowedTags=new Set(['BR','STRONG','B','EM','I','SPAN','MARK','P','DIV','UL','OL','LI','TABLE','THEAD','TBODY','TR','TH','TD','HR','CODE']);
  const allowedStyles=new Set([
    'color','background','background-color','font-weight','font-size','font-style','line-height',
    'padding','padding-top','padding-right','padding-bottom','padding-left',
    'margin','margin-top','margin-right','margin-bottom','margin-left',
    'border','border-top','border-right','border-bottom','border-left','border-color','border-width','border-style','border-radius',
    'text-align','white-space','letter-spacing','text-decoration'
  ]);
  [...template.content.querySelectorAll('*')].forEach(el=>{
    if(!allowedTags.has(el.tagName)){
      el.replaceWith(...el.childNodes);
      return;
    }
    [...el.attributes].forEach(attr=>{
      if(attr.name.toLowerCase()!=='style') el.removeAttribute(attr.name);
    });
    if(el.hasAttribute('style')){
      [...el.style].forEach(prop=>{
        const value=el.style.getPropertyValue(prop);
        if(!allowedStyles.has(prop)||/url\s*\(|expression\s*\(|javascript:|@import/i.test(value)){
          el.style.removeProperty(prop);
        }
      });
      if(!el.getAttribute('style')) el.removeAttribute('style');
    }
  });
  return template.innerHTML;
}

// 해설 HTML 안에서 마크다운 일부(굵게/기울임)만 변환. <br>은 이미 들어있다고 가정.
function renderExplainMarkdown(s){
  if(typeof s !== 'string') return '';
  let html = sanitizeExplainHtml(s).replace(/\*\*([^*]+)\*\*/g, '<strong>$1</strong>');
  html = html.replace(/(^|[^*])\*([^*\n]+)\*/g, '$1<em>$2</em>');

  var sectionColors = {
    '용어 정리':   {bg:'#EFF6FF', border:'#BFDBFE', color:'#1D4ED8'},
    '정답 해설':   {bg:'#F0FDF4', border:'#BBF7D0', color:'#15803D'},
    '핵심 암기':   {bg:'#FFFBEB', border:'#FDE68A', color:'#B45309'},
    '지문 독해':   {bg:'#EFF6FF', border:'#BFDBFE', color:'#1D4ED8'},
    '보기 해석':   {bg:'#F5F3FF', border:'#DDD6FE', color:'#6D28D9'},
    '왜 정답인가?':{bg:'#F0FDF4', border:'#BBF7D0', color:'#15803D'}
  };

  var sectionPattern = /━━\s*([^━]+?)\s*━━/g;
  var parts = [];
  var lastIndex = 0;
  var match;
  while((match = sectionPattern.exec(html)) !== null){
    if(match.index > lastIndex) parts.push({type:'text', content: html.slice(lastIndex, match.index)});
    parts.push({type:'section', name: match[1].trim()});
    lastIndex = match.index + match[0].length;
  }
  if(lastIndex < html.length) parts.push({type:'text', content: html.slice(lastIndex)});

  var result = '';
  var currentSection = null;
  var currentContent = '';

  function flushSection(){
    if(currentSection === null){
      if(currentContent.trim()) result += '<div style="font-size:14px;line-height:1.8;color:#0C4A6E">' + currentContent + '</div>';
    } else {
      var sc = sectionColors[currentSection] || {bg:'#F8FAFC', border:'#E2E8F0', color:'#334155'};
      result += '<div style="margin-bottom:10px;border-radius:10px;border:1.5px solid ' + sc.border + ';overflow:hidden">'
             + '<div style="padding:7px 13px;background:' + sc.border + ';font-size:12px;font-weight:800;color:' + sc.color + ';letter-spacing:0.5px">' + currentSection + '</div>'
             + '<div style="padding:11px 13px;background:' + sc.bg + ';font-size:13.5px;line-height:1.9;color:#1E293B">' + currentContent.replace(/^(<br>|\s)+/, '') + '</div>'
             + '</div>';
    }
    currentContent = '';
  }

  for(var i=0; i<parts.length; i++){
    var part = parts[i];
    if(part.type === 'section'){
      flushSection();
      currentSection = part.name;
    } else {
      currentContent += part.content;
    }
  }
  flushSection();
  return sanitizeExplainHtml(result);
}

// 문제 고유 ID: 년도|급코드|회차|과목|번호 — 세션 간 안정
function pqid(q){
  const short = q._short || q._gradeShort || '?';
  return `${q._year||'?'}|${short}|${q['회차']||'?'}|${q['과목']||'?'}|${q['번호']||'?'}`;
}
function pqSnapshot(q){
  return {
    _year: q._year, _short: q._short||q._gradeShort||'', _session: q._session,
    '회차': q['회차'], '과목': q['과목'], '번호': q['번호'],
    '문제': q['문제'], '선택지': q['선택지'], '정답': q['정답'],
  };
}
function normalizePastSavedMap(value){
  if(!value || typeof value!=='object' || Array.isArray(value)) return {map:{},changed:!!value};
  const map={};
  let changed=false;
  for(const [storedId,entry] of Object.entries(value)){
    if(!entry || typeof entry!=='object' || !entry.q){
      map[storedId]=entry;
      continue;
    }
    const exactId=pqid(entry.q);
    const normalized={...entry,id:exactId};
    const previous=map[exactId];
    if(!previous || Number(normalized.at||0)>=Number(previous.at||0)) map[exactId]=normalized;
    if(storedId!==exactId || entry.id!==exactId) changed=true;
  }
  return {map,changed};
}

function loadPastWrongs(){
  try{
    const raw=localStorage.getItem(PAST_WRONGS_KEY);
    if(raw){
      const normalized=normalizePastSavedMap(JSON.parse(raw));
      if(normalized.changed) savePastWrongs(normalized.map);
      return normalized.map;
    }
  }catch(e){console.warn('[v4.81] 기출 오답 노트 로드 실패:',e)}
  return {};
}
function savePastWrongs(m){
  return safeStorageSet(PAST_WRONGS_KEY,JSON.stringify(m),'기출 오답 노트');
}
function addPastWrong(q, myAns){
  const m = loadPastWrongs();
  const id = pqid(q);
  m[id] = { id, q: pqSnapshot(q), myAns, at: Date.now() };
  // 최대치 초과 시 오래된 것부터 제거
  const keys = Object.keys(m);
  if(keys.length > PAST_WRONGS_MAX){
    keys.sort((a,b)=>m[a].at - m[b].at);
    keys.slice(0, keys.length - PAST_WRONGS_MAX).forEach(k=>delete m[k]);
  }
  savePastWrongs(m);
}
function removePastWrong(id){
  const m = loadPastWrongs(); if(m[id]){ delete m[id]; savePastWrongs(m); }
}

  return {};
}
function savePastHistory(h){
  return safeStorageSet(PAST_HIST_KEY,JSON.stringify(h),'기출 풀이 기록');
}
function pastHistKey(baseShort, year, session){ return `${baseShort}|${year}|${session}`; }
function recordPastSession(baseShort, year, session, correct, total, mode){
  if(!baseShort || !year || !session || !total) return;
  const h = loadPastHistory();
  const key = pastHistKey(baseShort, year, session);
  const prev = h[key];
  const rate = correct/total;
  h[key] = {
    correct, total, mode, at: Date.now(),
    best: prev ? Math.max(prev.best||0, rate) : rate,
    tries: (prev?prev.tries:0)+1,
  };
  savePastHistory(h);
}
function getPastSessionHistory(baseShort, year, session){
  return loadPastHistory()[pastHistKey(baseShort, year, session)] || null;
}
