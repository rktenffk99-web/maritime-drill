}

// v4.34: short(변종) → 상선/어선/국내한정 구분 라벨 ('' 이면 구분 불필요)
// v4.45: engi5/engi5d → '일반'/'국내한정' 구분 추가
function variantLabel(short){
  if(/^navi\d/.test(short)){
    const m = short.match(/^navi\d(de|e|d)?$/);
    if(!m) return '';
    const suf = m[1] || '';
    if(suf === '') return '상선';
    if(suf === 'd') return '국내한정(상선)';
    if(suf === 'e') return '어선';
    if(suf === 'de') return '국내한정(어선)';
  }
  if(/^engi\d/.test(short)){
    if(short === 'engi5')  return '일반';
    if(short === 'engi5d') return '국내한정';
  }
  return '';
}
// v4.34: variant='all'|'상선'|'어선' 필터 통과 여부
// v4.45: engi5 계열용 '일반'/'국내한정' variant 추가
function variantMatches(short, variant){
  if(!variant || variant === 'all') return true;
  const lab = variantLabel(short);
  if(variant === '상선') return lab === '상선' || lab === '국내한정(상선)';
  if(variant === '어선') return lab === '어선' || lab === '국내한정(어선)';
  if(variant === '일반') return lab === '일반';
  if(variant === '국내한정') return lab === '국내한정';
  return true;
}
// v4.45: 현재 exams 풀에서 노출할 variant 필터 (title + 선택지)를 결정.
//   navi: '상선'/'어선' (국내한정은 각각에 합류)
//   engi5: '일반'/'국내한정'
//   필터가 불필요(변종 1종 이하)하면 choices=[]
function getVariantChoices(exams){
  const labs = [...new Set(exams.map(p => variantLabel(p.meta.gradeShort)).filter(Boolean))];
  const hasSS = labs.some(l => l.indexOf('상선')>=0);
  const hasYU = labs.some(l => l.indexOf('어선')>=0);
  if(hasSS && hasYU){
    return {title:'상선 / 어선', choices:['상선','어선']};
  }
  const hasNormal = labs.includes('일반');
  const hasRestrict = labs.includes('국내한정');
  if(hasNormal && hasRestrict){
    return {title:'일반 / 국내한정', choices:['일반','국내한정']};
  }
  return {title:'', choices:[]};
}

  });

function n3aPracticeGroups(data,source,id){

  };

  }else if(source==='topic'){
    const topic=(data.topics||[]).find(item=>item.id===id);
    const groupIds=new Set(n3aFilteredHits(topic&&topic.hits,yearSet).map(hit=>hit.groupId));

      (!query||`${entry.question} ${entry.subject}`.toLowerCase().includes(query)));
  }else if(navi3AnalysisState.view==='repeats'){
    entries=n3aBuildEntries(data);
  }else{
    const topicEntries=n3aBuildEntries(data);
    const groupIds=new Set(topicEntries.flatMap(topic=>topic.hits.map(hit=>hit.groupId)));
    entries=[...groupIds].map(groupId=>toEntry(groupsById.get(groupId))).filter(Boolean);
  }
  entries.sort((a,b)=>{
    if(navi3AnalysisState.sort==='count')return b.count-a.count||b.score-a.score||b.latest-a.latest;
