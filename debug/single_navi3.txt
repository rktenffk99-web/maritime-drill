// v4.45: engi5/engi5d → '일반'/'국내한정' 구분 추가
function variantLabel(short){
  if(/^navi\d/.test(short)){
    const m = short.match(/^navi\d(de|e|d)?$/);
    if(!m) return '';
    const suf = m[1] || '';
    if(suf === '') return '상선';
    if(suf === 'd') return '국내한정(상선)';
    if(suf === 'e') return '어선';
    if(suf === 'de') return '국내한정(어선)';
  }
  if(/^engi\d/.test(short)){
    if(short === 'engi5')  return '일반';
    if(short === 'engi5d') return '국내한정';
  }
  return '';
}

// v4.45: engi5 계열용 '일반'/'국내한정' variant 추가
function variantMatches(short, variant){
  if(!variant || variant === 'all') return true;
  const lab = variantLabel(short);
  if(variant === '상선') return lab === '상선' || lab === '국내한정(상선)';
  if(variant === '어선') return lab === '어선' || lab === '국내한정(어선)';
  if(variant === '일반') return lab === '일반';
  if(variant === '국내한정') return lab === '국내한정';
  return true;
}
