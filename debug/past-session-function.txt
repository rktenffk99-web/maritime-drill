async function startPastSession(mode, subjFilter){
  const curV = pastVariant || 'all';
  const exams = getPastExamsForSession(pastBaseShort, pastYear, pastSession, curV);
  if(!exams.length){ showToast('데이터를 찾을 수 없습니다.'); return; }

  }
  // 모든 변종 문제 합치기
  const allQuestions = [];
  exams.forEach(p => p.questions.forEach(q => allQuestions.push({
    ...q, _year: pastYear, _session: pastSession, '회차': pastSession, _short: p.meta.gradeShort
  })));
  // 과목 필터 결정
  let filterSet;
  if(!subjFilter || subjFilter === 'all'){
    filterSet = new Set(allQuestions.map(q=>q['과목']));
  } else if(Array.isArray(subjFilter)){
    filterSet = new Set(subjFilter);
  } else {
    filterSet = new Set([subjFilter]);
  }
  // 과목별로 모으기
  const bySubj = new Map();
  allQuestions.forEach(q => {
    if(!filterSet.has(q['과목'])) return;

    bySubj.get(q['과목']).push(q);
  });

    bySubj.forEach(arr => {
      const s = shuffle(arr.slice());
      pool.push(...s.slice(0, PAST_MOCK_PER_SUBJECT));
    });
    pool = shuffle(pool);
  } else {
    bySubj.forEach(arr => pool.push(...arr));

// v4.51: SM-2 기반 개인화 간격 반복
// 각 문제마다 난이도(ef)/다음 복습 간격(interval)/연속 정답(reps)/다음 복습일(nextReview) 추적
// Leitner 박스는 UI/통계 지표로 그대로 유지, 스케줄은 SM-2 interval이 주도
const SM2_DEFAULT_EF = 2.5;
const SM2_MIN_EF = 1.3;

// SM-2 품질 점수 매핑 (이 앱의 3버튼 입력):
// ○ 맞음(sure)   -> q=5 (perfect)
// △ 애매(unsure) -> q=3 (correct with difficulty)
// ✕ 틀림         -> q=1 (incorrect)
function qualityFromInput(correct, confidence){
  if(!correct) return 1;
  if(confidence==='sure') return 5;
  return 3;
}

// 순수 SM-2 계산. item을 변이시키고 nextReview(ISO) 갱신.
function updateSM2(item, q){
  // EF 갱신 (q에 무관하게 항상 적용되는 게 원안)
  let ef = (typeof item.ef==='number') ? item.ef : SM2_DEFAULT_EF;
  ef = ef + (0.1 - (5-q)*(0.08 + (5-q)*0.02));
  if(ef < SM2_MIN_EF) ef = SM2_MIN_EF;
  item.ef = Math.round(ef*1000)/1000;

  let reps = item.reps || 0;
  let interval;

  if(q < 3){
    // 실패 → 처음부터
    reps = 0;
    interval = 1;
  } else {
    if(reps === 0) interval = 1;
    else if(reps === 1) interval = 6;
    else interval = Math.round((item.interval||1) * item.ef);
    reps += 1;
  }

  item.reps = reps;
  item.interval = interval;
  const next = new Date();
  next.setHours(0,0,0,0);
  next.setDate(next.getDate() + interval);
  item.nextReview = next.toISOString();
}

// 같은 날 평가가 여러 개면 가장 낮은 품질을 사용해 과도한 복습 연장을 막습니다.

function updateSpacedReview(item,q){
  const today=studyDateKey();
  if(item.sm2Day!==today){
    item.sm2Day=today;
    item.sm2DayQuality=q;
    item.sm2DayBase={ef:item.ef,reps:item.reps,interval:item.interval,nextReview:item.nextReview};
    updateSM2(item,q);
    return;
  }
  const dailyQ=Math.min(typeof item.sm2DayQuality==='number'?item.sm2DayQuality:5,q);
  if(dailyQ===item.sm2DayQuality)return;
  const base=item.sm2DayBase||{};

  item.reps=typeof base.reps==='number'?base.reps:0;
  item.interval=typeof base.interval==='number'?base.interval:0;
  item.nextReview=base.nextReview||null;
  item.sm2DayQuality=dailyQ;
  updateSM2(item,dailyQ);
}

    if(typeof it.needsRetry!=='boolean') it.needsRetry=(it.wrong||0)>0&&(it.box||1)<3;

    if(!('lastRecoveryDate' in it)) it.lastRecoveryDate=null;
    // v4.84 이전 학습 기록은 첫 학습일을 알 수 없으므로 신규 목표에서 제외합니다.
    if(!('firstStudied' in it)) it.firstStudied=it.lastStudied?'legacy':null;
  });
  return loadedBoxes;
}

}

function isDue(item){
  if(!item) return false;
  // SM-2 경로: nextReview 있으면 그것만 본다
  if(item.nextReview){
    return new Date(item.nextReview).getTime() <= Date.now();
  }
  // 레거시 fallback: lastStudied + Leitner BOX_INT
  if(!item.lastStudied) return false;
  const d = (Date.now()-new Date(item.lastStudied).getTime())/(1e3*60*60*24);
  return d >= (BOX_INT[item.box]||1);
}
