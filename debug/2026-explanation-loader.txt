async function inflateBundledScript(payload){
  if(typeof DecompressionStream==='undefined'){
    throw new Error('Android Chrome 최신 버전이 필요합니다.');
  }
  const binary=atob(payload);
  const bytes=new Uint8Array(binary.length);
  for(let i=0;i<binary.length;i++)bytes[i]=binary.charCodeAt(i);
  const stream=new Blob([bytes]).stream().pipeThrough(new DecompressionStream('gzip'));
  return new Response(stream).text();
}
function runBundledScript(code,src){

}
function loadOptionalScript(src){
  if(optionalScriptLoads.has(src)) return optionalScriptLoads.get(src);
  const payload=getBundledOptionalScript(src);
  if(payload){
    const task=(async()=>{
      try{
        const code=await inflateBundledScript(payload);
        runBundledScript(code,src);
        return true;
      }catch(err){
        console.error('내장 데이터 로드 실패:',src,err);
        optionalScriptLoads.delete(src);
        return false;
      }
    })();
    optionalScriptLoads.set(src,task);
    return task;
  }
  const task=new Promise(resolve=>{
    const script=document.createElement('script');
    script.src=src;
    script.async=true;
    let settled=false;
    const finish=ok=>{
      if(settled) return;
      settled=true;
      clearTimeout(timer);
      if(!ok){ script.remove(); optionalScriptLoads.delete(src); }
      resolve(ok);
    };
    const timer=setTimeout(()=>finish(false),20000);
    script.onload=()=>finish(true);
    script.onerror=()=>finish(false);
    document.head.appendChild(script);
  });
  optionalScriptLoads.set(src,task);
  return task;
}

async function loadScriptsWithLimit(files, limit){
  const list=[...(files||[])];
  if(!list.length) return [];
  const results=new Array(list.length).fill(false);
  let next=0;
  async function worker(){
    while(next<list.length){
      const index=next++;
      results[index]=await loadOptionalScript(list[index]);
    }
  }
  const workers=Array.from({length:Math.min(Math.max(1,limit||1),list.length)},()=>worker());
  await Promise.all(workers);
  return results;
}

function variantLabel(short){
  if(/^navi\d/.test(short)){
    const m = short.match(/^navi\d(de|e|d)?$/);
    if(!m) return '';
    const suf = m[1] || '';
    if(suf === '') return '상선';
    if(suf === 'd') return '국내한정(상선)';
    if(suf === 'e') return '어선';
    if(suf === 'de') return '국내한정(어선)';
  }
  if(/^engi\d/.test(short)){
    if(short === 'engi5')  return '일반';
    if(short === 'engi5d') return '국내한정';
  }
  return '';
}
