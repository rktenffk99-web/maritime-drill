function ppTrainingYears(data){
    const metaYears=(data.meta&&data.meta.years||[]).map(Number);
    const hitYears=[];
    (data.groups||[]).forEach(group=>(group.hits||[]).forEach(hit=>{const y=Number(hit.year);if(Number.isFinite(y))hitYears.push(y)}));
    const all=[...new Set([...metaYears,...hitYears])].sort((a,b)=>a-b);
    // 2026 is the current exam year. Include it even when the source marks it as a partial year.
    if(all.includes(2026))return all.filter(y=>y<=2026).slice(-5);
    const partial=new Set((data.meta&&data.meta.partialYears||[]).map(Number));
    const complete=((data.meta&&data.meta.completeYears||[]).length?(data.meta.completeYears||[]).map(Number):all.filter(y=>!partial.has(y))).sort((a,b)=>a-b);
    return complete.slice(-5);
  }

  function ppUrgentSort(a,b,progress){
    const ra=ppProgressFor(progress,a.key),rb=ppProgressFor(progress,b.key);
    const wa=(ra.status==='weak'||ra.lastOutcome==='wrong'||ra.lastOutcome==='unsure')?1:0;
    const wb=(rb.status==='weak'||rb.lastOutcome==='wrong'||rb.lastOutcome==='unsure')?1:0;
    if(wb!==wa)return wb-wa;
    if(a.count!==b.count)return b.count-a.count;
    const da=ra.dueDate||'9999-12-31',db=rb.dueDate||'9999-12-31';
    if(da!==db)return da.localeCompare(db);
    return b.latest-a.latest;

  function ppAllocateNewSlots(requests,slots){
    const out={navi2:0,navi3:0};
    const active=PLAN_GRADES.filter(g=>requests[g]>0);
    if(slots<=0||!active.length)return out;
    const totalReq=active.reduce((s,g)=>s+requests[g],0);
    if(totalReq<=slots){active.forEach(g=>out[g]=requests[g]);return out}
    let used=0;
    active.forEach(g=>{out[g]=Math.min(requests[g],Math.floor(slots*requests[g]/totalReq));used+=out[g]});
    while(used<slots){
      const g=active.sort((a,b)=>(requests[b]-out[b])-(requests[a]-out[a]))[0];
      if(!g||out[g]>=requests[g])break;
      out[g]++;used++;
    }
    return out;
  }
  function ppBuildTodayAssignment(plan,pools,progress,today){
    const cap=Math.max(40,Math.min(250,Number(plan.dailyCap)||120));
    const allItems=PLAN_GRADES.flatMap(g=>(plan.grades[g].enabled&&plan.grades[g].examDate)?(pools[g]||[]):[]);
    const due=allItems.filter(item=>ppIsDue(ppProgressFor(progress,item.key),today)).sort((a,b)=>ppUrgentSort(a,b,progress));
    const dueSelected=due.slice(0,cap);
    let slots=Math.max(0,cap-dueSelected.length);
    const phases={},requests={navi2:0,navi3:0};
    PLAN_GRADES.forEach(g=>{
      if(!plan.grades[g].enabled||!plan.grades[g].examDate){phases[g]=null;return}
      const phase=ppPhaseForGrade(plan,g,pools[g]||[],progress,today);phases[g]=phase;
      const remaining=phase.candidates.filter(item=>!ppProgressFor(progress,item.key).firstPassDate);
      requests[g]=remaining.length?Math.ceil(remaining.length/Math.max(1,phase.learningDays)):0;
      if(phase.phase==='final'&&remaining.length)requests[g]=Math.ceil(remaining.length/Math.max(1,phase.dday||1));
    });
    const quotas=ppAllocateNewSlots(requests,slots);
    const newItems=[];
    PLAN_GRADES.forEach(g=>{
      const phase=phases[g];if(!phase||!quotas[g])return;
      const candidates=phase.candidates.filter(item=>!ppProgressFor(progress,item.key).firstPassDate&&!dueSelected.some(d=>d.key===item.key));
      candidates.sort((a,b)=>(ppHas2026(b)?1:0)-(ppHas2026(a)?1:0)||b.count-a.count||b.latest-a.latest);
      newItems.push(...candidates.slice(0,quotas[g]));
    });
    // 남는 슬롯이 있고 현재 단계에 미학습 문제가 더 있으면 시험이 더 가까운 급부터 채웁니다.
    slots=Math.max(0,cap-dueSelected.length-newItems.length);

        const phase=phases[g];if(!phase)return;
        phase.candidates.forEach(item=>{
          if(ppProgressFor(progress,item.key).firstPassDate)return;
          if(dueSelected.some(d=>d.key===item.key)||newItems.some(n=>n.key===item.key))return;
          extra.push(item);
        });
      });
      extra.sort((a,b)=>{

        const da=pa&&pa.dday!=null?pa.dday:999,db=pb&&pb.dday!=null?pb.dday:999;
        return da-db||(ppHas2026(b)?1:0)-(ppHas2026(a)?1:0)||b.count-a.count||b.latest-a.latest;
      });
      // 필요량보다 과도하게 앞당겨 풀지는 않도록 최대 절반만 보너스로 사용.
      const bonus=Math.min(slots,Math.ceil(cap*0.15));
      newItems.push(...extra.slice(0,bonus));
    }
    return {keys:[...dueSelected,...newItems].map(i=>i.key),createdAt:Date.now(),phases,requests,dueOverflow:Math.max(0,due.length-dueSelected.length),dueCount:dueSelected.length,newCount:newItems.length};

  function ppUrgentSort(a,b,progress){

  function ppAllocateNewSlots(requests,slots){
    const out={navi2:0,navi3:0};
    const active=PLAN_GRADES.filter(g=>requests[g]>0);
    if(slots<=0||!active.length)return out;
    const totalReq=active.reduce((s,g)=>s+requests[g],0);
    if(totalReq<=slots){active.forEach(g=>out[g]=requests[g]);return out}
    let used=0;
    active.forEach(g=>{out[g]=Math.min(requests[g],Math.floor(slots*requests[g]/totalReq));used+=out[g]});
    while(used<slots){
      const g=active.sort((a,b)=>(requests[b]-out[b])-(requests[a]-out[a]))[0];
      if(!g||out[g]>=requests[g])break;
      out[g]++;used++;
    }
    return out;
  }

function getPastExamsByBase(baseShort, variant){
  if(!window.MD_PAST) return [];
  return Object.values(window.MD_PAST).filter(p => {
    if(!p.meta) return false;
    if(baseFromShort(p.meta.gradeShort) !== baseShort) return false;
    return variantMatches(p.meta.gradeShort, variant);
  });
}
function getPastExam(short, year, session){
  if(!window.MD_PAST) return null;
  const id = `${year}-${short}-${session}`;
  return window.MD_PAST[id] || null;
}

function n3aEnsureFilterState(data){
  const state=navi3AnalysisState;
  const subjects=(data&&data.subjects||[]).map(String);
  const years=(data&&data.meta&&data.meta.years||[]).map(Number).sort((a,b)=>a-b);
  if(!state._filterLoaded){
    const saved=safeStorageGetJSON(n3aFilterStorageKey(),{},'빈출 필터');
    state.subjects=Array.isArray(saved.subjects)?saved.subjects.map(String):subjects.slice();
    state.years=Array.isArray(saved.years)?saved.years.map(Number):years.slice();
    state.minCount=[1,2,3,4].includes(Number(saved.minCount))?Number(saved.minCount):2;
    state._filterLoaded=true;
  }
  if(!Array.isArray(state.subjects))state.subjects=subjects.slice();
  if(!Array.isArray(state.years))state.years=years.slice();
  state.subjects=state.subjects.filter(subject=>subjects.includes(subject));
  state.years=state.years.map(Number).filter(year=>years.includes(year));
  if(![1,2,3,4].includes(Number(state.minCount)))state.minCount=2;
  return state;
}

  return Math.min(100,Math.round(Math.min(60,hits.length/7*60)+n3aRecencyScore(years[years.length-1],maxYear)+Math.min(15,years.length/6*15)));
}
function n3aTopicScore(hits,yearSet,totalSessions){
  const years=n3aUniqueYears(hits),latest=years[years.length-1];
  const sessions=new Set(hits.map(n3aSessionKey)).size;
  const coverage=totalSessions?Math.min(1,sessions/totalSessions):0;
  const density=totalSessions?Math.min(1,hits.length/(totalSessions*4)):0;
  return Math.min(100,Math.round(coverage*55+density*20+n3aRecencyScore(latest,Math.max(...yearSet))));
}
function n3aAllSessionKeys(data,yearSet){
  const keys=new Set();
  (data.groups||[]).forEach(group=>n3aFilteredHits(group.hits,yearSet).forEach(hit=>keys.add(n3aSessionKey(hit))));
  return keys;
}
function n3aRangeStats(data,yearSet){
  let questions=0,repeated=0;
  const minCount=Number(navi3AnalysisState.minCount)||2;
  (data.groups||[]).forEach(group=>{
    if(!n3aSubjectSelected(group.subject))return;
    const count=n3aFilteredHits(group.hits,yearSet).length;
    questions+=count;
    if(count>=minCount)repeated+=1;
  });
  return {questions,repeated,sessions:n3aAllSessionKeys(data,yearSet).size};
}
function n3aBuildEntries(data){
  const state=navi3AnalysisState;
  const yearSet=n3aRangeYears(data,state.range);
  const query=String(state.query||'').trim().toLowerCase();
  const totalSessions=n3aAllSessionKeys(data,yearSet).size;
  let entries=[];
  if(state.view==='topics'){
    entries=(data.topics||[]).map(topic=>{
      const hits=n3aFilteredHits(topic.hits,yearSet);
      const years=n3aUniqueYears(hits);
      const sessions=new Set(hits.map(n3aSessionKey));
      return {...topic,hits,years,latest:years[years.length-1]||0,sessionCount:sessions.size,occurrences:hits.length,distinctQuestions:new Set(hits.map(hit=>hit.groupId)).size,score:hits.length?n3aTopicScore(hits,yearSet,totalSessions):0};
    }).filter(entry=>entry.hits.length&&
      n3aSubjectSelected(entry.subject)&&
      (!query||`${entry.label} ${entry.subject}`.toLowerCase().includes(query)));
  }else{
    entries=(data.groups||[]).map(group=>{
      const hits=n3aFilteredHits(group.hits,yearSet);
      const years=n3aUniqueYears(hits);
      return {...group,hits,years,count:hits.length,latest:years[years.length-1]||0,score:hits.length?n3aRepeatScore(hits,yearSet):0};
    }).filter(entry=>entry.count>=(Number(state.minCount)||2)&&
      n3aSubjectSelected(entry.subject)&&
      (!query||`${entry.question} ${entry.subject}`.toLowerCase().includes(query)));
  }
  entries.sort((a,b)=>{
    if(state.sort==='count')return (b.occurrences||b.count)-(a.occurrences||a.count)||b.score-a.score;
    if(state.sort==='recent')return b.latest-a.latest||b.score-a.score;
    return b.score-a.score||(b.occurrences||b.count)-(a.occurrences||a.count);
  });
  return entries;
}

function n3aPracticeGroups(data,source,id){
  const yearSet=n3aRangeYears(data,navi3AnalysisState.range);
  const groupsById=new Map((data.groups||[]).map(group=>[group.id,group]));
  const toEntry=group=>{
    if(!group)return null;
    const hits=n3aFilteredHits(group.hits,yearSet);
    const years=n3aUniqueYears(hits);
    return hits.length?{...group,hits,years,count:hits.length,latest:years[years.length-1]||0,score:n3aRepeatScore(hits,yearSet)}:null;
  };
  let entries=[];
  if(source==='group'){
    entries=[toEntry(groupsById.get(id))].filter(Boolean);
  }else if(source==='topic'){
    const topic=(data.topics||[]).find(item=>item.id===id);
    const groupIds=new Set(n3aFilteredHits(topic&&topic.hits,yearSet).map(hit=>hit.groupId));
    entries=[...groupIds].map(groupId=>toEntry(groupsById.get(groupId))).filter(Boolean);
  }else if(source==='filtered'){
    const query=String(navi3AnalysisState.query||'').trim().toLowerCase();
    entries=(data.groups||[]).map(toEntry).filter(entry=>entry&&
      entry.count>=(Number(navi3AnalysisState.minCount)||2)&&
      n3aSubjectSelected(entry.subject)&&
      (!query||`${entry.question} ${entry.subject}`.toLowerCase().includes(query)));
  }else if(navi3AnalysisState.view==='repeats'){
    entries=n3aBuildEntries(data);
  }else{
    const topicEntries=n3aBuildEntries(data);
    const groupIds=new Set(topicEntries.flatMap(topic=>topic.hits.map(hit=>hit.groupId)));
    entries=[...groupIds].map(groupId=>toEntry(groupsById.get(groupId))).filter(Boolean);
  }
  entries.sort((a,b)=>{
    if(navi3AnalysisState.sort==='count')return b.count-a.count||b.score-a.score||b.latest-a.latest;
    if(navi3AnalysisState.sort==='recent')return b.latest-a.latest||b.score-a.score||b.count-a.count;
    return b.score-a.score||b.count-a.count||b.latest-a.latest;
  });
  return entries;
}
function n3aFindQuestion(group,hit){

}
